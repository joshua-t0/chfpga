  771 -- ctrl_reset pulse generator

  773 -- Generates a long pulse on ctrl_reset_pulse when a low-to-high transition is seen on ctrl_reset_trig.
  774 -- Operates on ctrl_clk.
  775 ctrl_reset_blk: block
  776         signal ctrl_reset_trig_reg: std_logic;
  777         signal ctrl_reset_ctr: unsigned(3 downto 0);
  778 begin
  779         ctrl_reset_proc: process(ctrl_clk)
  780         begin
  781                 if rising_edge(ctrl_clk) then
  782                         ctrl_reset_trig_reg <= ctrl_reset_trig;
  783 
  784                         if signed(ctrl_reset_ctr) = -1 then
  785                                 ctrl_reset_pulse <= '0';
  786                         else
  787                                 ctrl_reset_ctr <= ctrl_reset_ctr + 1;
  788                         end if;
  789 
  790                         if ctrl_reset_trig='1' and ctrl_reset_trig_reg='0' then
  791                                 ctrl_reset_pulse <= '1';
  792                                 ctrl_reset_ctr <= (others => '0');
  793                         end if;
  794 
  795                 end if; --clk
  796         end process;
  797 end block;

 1026 -- IRIG-B Time capture
 1027 -----------------------
 1028 --
 1029 -- This process is used to capture the IRIG-B time and count the PPS pulses.
 1030 -- IRIG-B capture is triggered either by capturing the reference clock counter
 1031 -- or by capturing the event/frame event.
 1032 --
 1033 -- `event_ts_sample_trig` is handled by the event/frame capture logic, which operates
 1034 -- on on the event_ctr_clk (chan_clk(0), 200 MHz derived from the ADC or
 1035 -- multi-phase PLL).
 1036 --
 1037 --
 1038 -- ts_sample_trig10 is handled by the refclk counter capture logic, which is
 1039 -- clocked on clk10.
 1040 --
 1041 -- This process operates in the clk200 clock domain. Ultimately all the
 1042 -- trigger clocks are derived from clk10, but they have an unknown phase. So
 1043 -- he have to deal with clock domain crossing on the trigger.
 1044 --
 1045 -- Triggering is done by inverting the state of `event_ts_sample_trig` or
 1046 -- `ts_sample_trig10`. The capture is done when  the signal differ from the
 1047 -- correspnding acknowledge bit. The acknowledge bit is then set to the
 1048 -- trigger signal to indicate that the caapture is done.
 1049 --
 1050 -- Note that the acknowledge signals are on clk200, not on the clock of their
 1051 -- corresponding trigger.
 1052 --
